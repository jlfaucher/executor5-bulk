ooRexx> codepoint2 = .RexxUnicodeServices~utf8DecodePreviousCodepoint(string, indexB, >size)

 4
ooRexx> indexB -= abs(size); indexB=                                                            -- 31

ooRexx> codepoint1 = .RexxUnicodeServices~utf8DecodePreviousCodepoint(string, indexB, >size)
ooRexx> codepoint1~d2x=; size=                                                                  -- '1F467'; 4
'1F467'
 4
ooRexx> indexB -= abs(size); indexB=                                                            -- 27

'MICR ON US SYMBOL'
ooRexx> .RexxUnicodeCharacter~new("D800"~x2d)~nameAlias= -- ''
''


-- The special value -1 has no name alias
ooRexx> .RexxUnicodeCharacter~new(-1)~nameAlias=         -- ''
''


--------------------------------------------------------------------------------
-- aRexxUnicodeCharacter~string
--------------------------------------------------------------------------------

ooRexx> .RexxUnicodeCharacter~new(2448~x2d)~string=     -- '"⑈" \x{E29188} U+2448 So Other_Symbol "OCR DASH"'
'"⑈" \x{E29188} U+2448 So Other_Symbol "OCR DASH"'
ooRexx> .RexxUnicodeCharacter~new("D800"~x2d)~string=   -- '<?> \x{EDA080} U+D800 Cs Surrogate "<lead surrogate-D800>"'
'<?> \x{EDA080} U+D800 Cs Surrogate "<lead surrogate-D800>"'


-- The special codepoint value -1 has its own string representation
ooRexx> .RexxUnicodeCharacter~new(-1)~string=         -- 'An invalid character'
'An invalid character'
